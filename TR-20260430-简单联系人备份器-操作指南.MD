```swift
import Contacts
import Foundation

final class ContactService {
    static let shared = ContactService()
    private let store = CNContactStore()

    func requestAccess() async -> Bool {
        do {
            return try await store.requestAccess(for: .contacts)
        } catch {
            print("Contact access denied: \(error)")
            return false
        }
    }

    func fetchAllContacts() async throws -> [CNContact] {
        let keys: [CNKeyDescriptor] = [
            CNContactIdentifierKey as CNKeyDescriptor,
            CNContactGivenNameKey as CNKeyDescriptor,
            CNContactFamilyNameKey as CNKeyDescriptor,
            CNContactOrganizationNameKey as CNKeyDescriptor,
            CNContactPhoneNumbersKey as CNKeyDescriptor,
            CNContactEmailAddressesKey as CNKeyDescriptor,
            CNContactPostalAddressesKey as CNKeyDescriptor,
            CNContactUrlAddressesKey as CNKeyDescriptor,
            CNContactSocialProfilesKey as CNKeyDescriptor,
            CNContactInstantMessageAddressesKey as CNKeyDescriptor,
            CNContactBirthdayKey as CNKeyDescriptor,
            CNContactDatesKey as CNKeyDescriptor,
            CNContactNoteKey as CNKeyDescriptor,
            CNContactImageDataKey as CNKeyDescriptor,
            CNContactThumbnailImageDataKey as CNKeyDescriptor,
            CNContactJobTitleKey as CNKeyDescriptor,
            CNContactDepartmentNameKey as CNKeyDescriptor,
            CNContactPrefixKey as CNKeyDescriptor,
            CNContactSuffixKey as CNKeyDescriptor,
            CNContactMiddleNameKey as CNKeyDescriptor,
            CNContactNicknameKey as CNKeyDescriptor,
            CNContactPhoneticGivenNameKey as CNKeyDescriptor,
            CNContactPhoneticFamilyNameKey as CNKeyDescriptor,
            CNContactPhoneticMiddleNameKey as CNKeyDescriptor,
            CNContactRelationsKey as CNKeyDescriptor,
            CNContactTypeKey as CNKeyDescriptor
        ]

        let request = CNContactFetchRequest(keysToFetch: keys)
        var contacts: [CNContact] = []

        try store.enumerateContacts(with: request) { contact, _ in
            contacts.append(contact)
        }

        return contacts.sorted { 
            ($0.familyName + $0.givenName) < ($1.familyName + $1.givenName) 
        }
    }

    func getContactCount() async throws -> Int {
        let contacts = try await fetchAllContacts()
        return contacts.count
    }

    func fetchContactsFromContainer(containerIdentifier: String?) async throws -> [CNContact] {
        let keys: [CNKeyDescriptor] = [
            CNContactIdentifierKey as CNKeyDescriptor,
            CNContactGivenNameKey as CNKeyDescriptor,
            CNContactFamilyNameKey as CNKeyDescriptor,
            CNContactPhoneNumbersKey as CNKeyDescriptor,
            CNContactEmailAddressesKey as CNKeyDescriptor
        ]

        let request = CNContactFetchRequest(keysToFetch: keys)
        if let containerId = containerIdentifier {
            request.containerIdentifiers = [containerId]
        }

        var contacts: [CNContact] = []
        try store.enumerateContacts(with: request) { contact, _ in
            contacts.append(contact)
        }
        return contacts
    }

    func getAvailableContainers() throws -> [CNContainer] {
        return try store.containers(matching: nil)
    }
}
```

### 4.3 VCF生成器（VCFGenerator.swift）— 严格RFC 6350规范

```swift
import Foundation
import Contacts

final class VCFGenerator {
    static func generate(contacts: [CNContact]) -> Data? {
        var vcardString = ""

        for contact in contacts {
            vcardString += vCardString(for: contact)
        }

        return vcardString.data(using: .utf8)
    }

    static func vCardString(for contact: CNContact) -> String {
        var lines: [String] = []
        lines.append("BEGIN:VCARD")
        lines.append("VERSION:3.0")
        lines.append("PRODID:-//ContactVault//EN")
        lines.append("UID:\(escapeVCF(contact.identifier))")

        if !contact.givenName.isEmpty || !contact.familyName.isEmpty {
            var nameComponents: [String] = []
            nameComponents.append(escapeVCF(contact.familyName))
            nameComponents.append(escapeVCF(contact.givenName))
            nameComponents.append(escapeVCF(contact.middleName))
            nameComponents.append(escapeVCF(contact.prefix))
            nameComponents.append(escapeVCF(contact.suffix))
            lines.append("N:\(nameComponents.joined(separator: ";"))")
            lines.append("FN:\(escapeVCF(contact.formattedName)))")
        }

        if !contact.organizationName.isEmpty {
            lines.append("ORG:\(escapeVCF(contact.organizationName))")
        }
        if !contact.jobTitle.isEmpty {
            lines.append("TITLE:\(escapeVCF(contact.jobTitle))")
        }
        if !contact.departmentName.isEmpty {
            lines.append("ROLE:\(escapeVCF(contact.departmentName))")
        }

        for phone in contact.phoneNumbers {
            let label = mapPhoneLabel(phone.label)
            let number = escapeVCF(phone.value.stringValue)
            lines.append("TEL;TYPE=\(label):\(number)")
        }

        for email in contact.emailAddresses {
            let label = mapEmailLabel(email.label)
            let address = escapeVCF(email.value as String)
            lines.append("EMAIL;TYPE=\(label):\(address)")
        }

        for address in contact.postalAddresses {
            let label = mapAddressLabel(address.label)
            let value = address.value
            let street = escapeVCF(value.street)
            let city = escapeVCF(value.city)
            let state = escapeVCF(value.state)
            let postalCode = escapeVCF(value.postalCode)
            let country = escapeVCF(value.country)
            lines.append("ADR;TYPE=\(label):;;\(street);\(city);\(state);\(postalCode);\(country)")
        }

        for url in contact.urlAddresses {
            let urlString = escapeVCF(url.value as String)
            lines.append("URL:\(urlString)")
        }

        if let birthday = contact.birthday {
            if let year = birthday.year, let month = birthday.month, let day = birthday.day {
                lines.append("BDAY:\(String(format: "%04d%02d%02d", year, month, day))")
            } else if let month = birthday.month, let day = birthday.day {
                lines.append("BDAY:--\(String(format: "%02d%02d", month, day))")
            }
        }

        if !contact.note.isEmpty {
            lines.append("NOTE:\(escapeVCF(contact.note))")
        }

        if let imageData = contact.thumbnailImageData {
            let base64 = imageData.base64EncodedString(options: [.lineLength64Characters, .endLineWithCarriageReturn])
            lines.append("PHOTO;ENCODING=BASE64;TYPE=JPEG:\(base64)")
        }

        lines.append("END:VCARD")
        return lines.joined(separator: "\r\n") + "\r\n"
    }

    static func escapeVCF(_ string: String) -> String {
        return string
            .replacingOccurrences(of: "\\", with: "\\\\")
            .replacingOccurrences(of: ",", with: "\\,")
            .replacingOccurrences(of: ";", with: "\\;")
            .replacingOccurrences(of: "\n", with: "\\n")
    }

    static func mapPhoneLabel(_ label: String?) -> String {
        guard let label = label else { return "CELL" }
        if label == CNLabelPhoneNumberMain { return "MAIN" }
        if label == CNLabelPhoneNumberMobile { return "CELL" }
        if label == CNLabelPhoneNumberHomeFax { return "HOME,FAX" }
        if label == CNLabelPhoneNumberWorkFax { return "WORK,FAX" }
        if label == CNLabelHome { return "HOME" }
        if label == CNLabelWork { return "WORK" }
        if label == CNLabelOther { return "OTHER" }
        return "CELL"
    }

    static func mapEmailLabel(_ label: String?) -> String {
        guard let label = label else { return "INTERNET" }
        if label == CNLabelHome { return "HOME,INTERNET" }
        if label == CNLabelWork { return "WORK,INTERNET" }
        return "INTERNET"
    }

    static func mapAddressLabel(_ label: String?) -> String {
        guard let label = label else { return "HOME" }
        if label == CNLabelHome { return "HOME" }
        if label == CNLabelWork { return "WORK" }
        return "OTHER"
    }
}
```

### 4.4 CSV生成器（CSVGenerator.swift）

```swift
import Foundation
import Contacts

final class CSVGenerator {
    static func generate(contacts: [CNContact]) -> Data? {
        let header = "First Name,Last Name,Organization,Job Title,Phone,Email,Street,City,State,Zip,Country,Note"
        var rows = [header]

        for contact in contacts {
            let phones = contact.phoneNumbers.map { $0.value.stringValue }.joined(separator: " | ")
            let emails = contact.emailAddresses.map { $0.value as String }.joined(separator: " | ")

            var addresses: [String] = []
            for addr in contact.postalAddresses {
                addresses.append("\(addr.value.street), \(addr.value.city)")
            }

            let row = [
                escapeCSV(contact.givenName),
                escapeCSV(contact.familyName),
                escapeCSV(contact.organizationName),
                escapeCSV(contact.jobTitle),
                escapeCSV(phones),
                escapeCSV(emails),
                escapeCSV(contact.postalAddresses.first?.value.street ?? ""),
                escapeCSV(contact.postalAddresses.first?.value.city ?? ""),
                escapeCSV(contact.postalAddresses.first?.value.state ?? ""),
                escapeCSV(contact.postalAddresses.first?.value.postalCode ?? ""),
                escapeCSV(contact.postalAddresses.first?.value.country ?? ""),
                escapeCSV(contact.note)
            ].joined(separator: ",")

            rows.append(row)
        }

        let csvContent = rows.joined(separator: "\n")
        return csvContent.data(using: .utf8)
    }

    static func escapeCSV(_ string: String) -> String {
        if string.contains(",") || string.contains("\"") || string.contains("\n") {
            return "\"" + string.replacingOccurrences(of: "\"", with: "\"\"") + "\""
        }
        return string
    }
}
```

### 4.5 加密服务（CryptoService.swift）

```swift
import CryptoKit
import Foundation
import LocalAuthentication

final class CryptoService {
    static let shared = CryptoService()

    func encrypt(data: Data, withPassword password: String) throws -> Data {
        let salt = generateSalt()
        let key = deriveKey(from: password, salt: salt)
        let nonce = AES.GCM.Nonce()
        let sealed = try AES.GCM.seal(data, using: key, nonce: nonce)

        var result = Data()
        result.append(salt)
        result.append(nonce.withUnsafeBytes { Data($0) })
        result.append(sealed.ciphertext)
        result.append(sealed.tag.withUnsafeBytes { Data($0) })

        return result
    }

    func decrypt(data: Data, withPassword password: String) throws -> Data {
        let salt = data.prefix(16)
        let nonceData = data.subdata(in: 16..<28)
        let tagData = data.suffix(16)
        let ciphertext = data.subdata(in: 28..<(data.count - 16))

        let key = deriveKey(from: password, salt: salt)
        let nonce = try AES.GCM.Nonce(data: nonceData)
        let tag = AES.GCM.Tag(data: tagData)
        let sealedBox = try AES.GCM.SealedBox(nonce: nonce, ciphertext: ciphertext, tag: tag)

        return try AES.GCM.open(sealedBox, using: key)
    }

    private func deriveKey(from password: String, salt: Data) -> SymmetricKey {
        let passwordData = password.data(using: .utf8)!
        let derivedKey = HKDF<SHA256>.deriveKey(
            inputKeyMaterial: SymmetricKey(data: passwordData),
            salt: salt,
            info: Data("ContactVault".utf8),
            outputByteCount: 32
        )
        return derivedKey
    }

    private func generateSalt() -> Data {
        var salt = Data(count: 16)
        salt.withUnsafeMutableBytes { saltBytes in
            _ = SecRandomCopyBytes(kSecRandomDefault, 16, saltBytes.baseAddress!)
        }
        return salt
    }

    func authenticateWithBiometrics() async -> Bool {
        let context = LAContext()
        var error: NSError?

        guard context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error) else {
            return false
        }

        do {
            return try await context.evaluatePolicy(
                .deviceOwnerAuthenticationWithBiometrics,
```

### 4.6 导出服务（ExportService.swift）

```swift
import Foundation
import Contacts
import UniformTypeIdentifiers

enum ExportFormat: String, CaseIterable {
    case vcf = "VCF"
    case csv = "CSV"

    var utType: UTType {
        switch self {
        case .vcf: return .vCard
        case .csv: return UTType(filenameExtension: "csv") ?? .plainText
        }
    }

    var fileExtension: String {
        switch self {
        case .vcf: return "vcf"
        case .csv: return "csv"
        }
    }
}

final class ExportService {
    static let shared = ExportService()

    private let fileManager = FileManager.default

    var backupsDirectory: URL {
        let appSupport = fileManager.urls(for: .applicationSupportDirectory, in: .userDomainMask).first!
        let dir = appSupport.appendingPathComponent("ContactVault/Backups", isDirectory: true)
        try? fileManager.createDirectory(at: dir, withIntermediateDirectories: true)
        return dir
    }

    func export(
        contacts: [CNContact],
        format: ExportFormat,
        encrypted: Bool = false,
        password: String? = nil
    ) async throws -> URL {
        let data: Data

        switch format {
        case .vcf:
            guard let vcfData = VCFGenerator.generate(contacts: contacts) else {
                throw ExportError.generationFailed
            }
            data = vcfData
        case .csv:
            guard let csvData = CSVGenerator.generate(contacts: contacts) else {
                throw ExportError.generationFailed
            }
            data = csvData
        }

        let timestamp = DateFormatter.backupFormatter.string(from: Date())
        let fileName = "ContactVault_\(timestamp)"

        let finalData: Data
        let fileExtension: String

        if encrypted, let pwd = password {
            finalData = try CryptoService.shared.encrypt(data: data, withPassword: pwd)
            fileExtension = "\(format.fileExtension).encrypted"
        } else {
            finalData = data
            fileExtension = format.fileExtension
        }

        let fileURL = backupsDirectory.appendingPathComponent("\(fileName).\(fileExtension)")
        try finalData.write(to: fileURL)

        let record = BackupRecord(
            id: UUID(),
            fileName: "\(fileName).\(fileExtension)",
            fileURL: fileURL,
            format: format,
            contactCount: contacts.count,
            isEncrypted: encrypted,
            fileSize: finalData.count,
            createdAt: Date()
        )
        BackupStore.shared.addRecord(record)

        return fileURL
    }

    func deleteBackup(_ record: BackupRecord) throws {
        try fileManager.removeItem(at: record.fileURL)
        BackupStore.shared.removeRecord(record)
    }
}

enum ExportError: LocalizedError {
    case generationFailed
    case encryptionFailed

    var errorDescription: String? {
        switch self {
        case .generationFailed: return "Failed to generate backup file"
        case .encryptionFailed: return "Failed to encrypt backup file"
        }
    }
}

extension DateFormatter {
    static let backupFormatter: DateFormatter = {
        let f = DateFormatter()
        f.dateFormat = "yyyy-MM-dd_HHmmss"
        return f
    }()
}
```

### 4.9 编写规则

1. **命名规范**：
   - 文件名：PascalCase，与主类型名一致
   - 类型名：PascalCase（`ContactService`, `BackupViewModel`）
   - 函数名：camelCase（`fetchAllContacts()`, `requestAccess()`）
   - 常量：PascalCase（`ContactVault.backupDirectory`）
   - 枚举case：camelCase（`.vcf`, `.csv`, `.weekly`）

2. **架构规范**：
   - MVVM架构：View → ViewModel → Service/Model
   - Service层单例模式（`ContactService.shared`）
   - ViewModel使用`@StateObject`/`@ObservableObject`
   - 异步操作使用`async/await`，不使用Combine
   - 错误处理使用`throws`，不使用Result类型

3. **SwiftUI规范**：
   - 使用`NavigationStack`（非NavigationView）
   - 使用`@StateObject`创建，`@ObservedObject`传入
   - 使用`task`修饰符替代`onAppear`进行异步操作
   - 使用`.ultraThinMaterial`实现毛玻璃效果
   - 使用SFSymbols图标，不使用自定义图标

4. **数据安全规范**：
   - 所有备份文件默认存储在App Sandbox
   - 加密备份使用AES-GCM-256
   - 密码派生使用HKDF-SHA256
   - 不存储用户密码，仅用于加密/解密时输入
   - Face ID/Touch ID用于解锁加密备份

5. **VCF规范**：
   - 严格遵循RFC 6350标准
   - 行尾使用CRLF（\r\n）
   - 特殊字符必须转义（逗号、分号、反斜杠、换行）
   - 版本声明为3.0（最大兼容性）
   - 不输出空字段（避免(null)值问题）

### 8.2 策略对比分析

| 策略 | 优势 | 劣势 | 适合场景 | 本项目适配度 |
|------|------|------|---------|------------|
| 完全免费 | 下载量最大 | 无收入，无法持续 | 引流工具 | ❌ 不适合 |
| **付费购买** | **零摩擦体验，用户信任，无广告** | **下载门槛** | **简单工具App** | **✅ 最适合** |
| 订阅A(免费+买断) | 用户体验好 | 免费版功能受限 | 中等功能App | ⚠️ 可选 |
| 订阅B(月/年/买断) | 收入上限高 | 订阅疲劳，用户反感 | 复杂App | ❌ 不适合 |

### 8.3 为什么选择付费购买策略

1. **市场定位分析**：
   - ContactVault是简单工具App，功能明确单一
   - 用户对"备份联系人"有明确付费意愿（$0.99~$6.99区间）
   - 竞品中Contacts Backup Utility $5.99付费购买，5个评分但4.6★
   - 竞品中Contacts Backup + Transfer $2.99 Pro，8.6K评分4.6★
   - **关键洞察**：用户愿意为"简单+隐私+无订阅"付费

2. **定价心理学**：
   - $2.99是"冲动购买"价格点（< $5心理门槛）
   - 比竞品订阅（$4.99/周）便宜52倍
   - "One-time purchase, yours forever" 是最强卖点
   - 无隐藏费用 = 信任 = 口碑传播

3. **收入模型测算**：

| 指标 | 保守估计 | 中等估计 | 乐观估计 |
|------|---------|---------|---------|
| 月下载量 | 500 | 2,000 | 5,000 |
| 转化率（付费下载） | 100% | 100% | 100% |
| 单价 | $2.99 | $2.99 | $2.99 |
| Apple分成后(70%) | $2.09 | $2.09 | $2.09 |
| 月收入 | $1,045 | $4,180 | $10,450 |
| 年收入 | $12,540 | $50,160 | $125,400 |

4. **竞品定价对比**：

| 竞品 | 定价模式 | 年费用 | ContactVault节省 |
|------|---------|--------|-----------------|
| Contacts Backup Pro | 订阅$4.99/月 | $59.88 | 95% |
| Contact.s Back Up | 订阅$49.99/年 | $49.99 | 94% |
| Contacts Backup+ | 一次性$2.99 | $2.99 | 0%（同等） |
| Contacts Backup Utility | 一次性$5.99 | $5.99 | 50% |

5. **定价策略细节**：

```
最终定价：$2.99（一次性购买）

定价理由：
├── 价格点分析
│   ├── $0.99 → 太低，暗示低质量
│   ├── $1.99 → 可选，但利润太薄
│   ├── $2.99 → ✅ 最佳平衡点（冲动购买+合理利润）
│   ├── $3.99 → 可选，但开始有犹豫
│   └── $5.99 → 太高，与免费竞品差距大
│
├── 价值锚定
│   ├── "Less than a cup of coffee" → $2.99 < $4.99咖啡
│   ├── "No subscription, ever" → 对比$59.88/年
│   └── "One-time $2.99" → 强调一次性
│
├── App Store展示策略
│   ├── Price: $2.99
│   ├── 无In-App Purchases → 信任信号
│   ├── 无广告 → 信任信号
│   └── 无订阅 → 信任信号
│
└── 促销策略
    ├── 首月上线：$1.99 限时优惠
    ├── 节假日：$0.99 促销
    └── 版本更新：恢复$2.99
```

### 8.4 为什么不选择订阅制

1. **用户反感**：App Store评论中大量"为什么简单的备份功能需要订阅？"
2. **功能简单**：联系人备份是一次性操作，不是持续服务，订阅不合理
3. **无API成本**：所有操作本地完成，无服务器/API费用
4. **竞品差异化**：所有主要竞品都是订阅制，一次性购买是最大差异化
5. **信任建立**：一次性购买 = 开发者不靠持续收费 = 用户体验优先

### 8.5 未来定价调整策略

| 阶段 | 定价 | 原因 |
|------|------|------|
| 上线1-3月 | $1.99 | 低价获取初始用户和评价 |
| 稳定期 | $2.99 | 标准定价 |
| 重大功能更新后 | $3.99 | 价值提升（如：重复联系人合并） |
| 促销期 | $0.99 | 节假日/里程碑促销 |

- Swift实现的联系人备份
- 前后端分离架构参考

**二次开发方向**：
